class Heuristic:
    def get_heuristic(self, goal):
        for node in self.adjacency_List:
            if node == goal:
                self.h[node] = 0
            else:
                self.h[node] = int(input(f"h({node}) = "))

    def a_star(self, start, goal):
        if start not in self.adjacency_List or goal not in self.adjacency_List:
          print("Invalid start or goal")
          return None
        self.get_heuristic(goal)
        frontier = [([start], self.h[start])]
        explored = []
        i = 1

        print("\nA* Traversal Table:")
        print("Iter No | Frontier                                                  | Explored")
        print("-" * 110)

        while frontier:
            frontier_list = [f"{path[-1]}({cost})" for path, cost in frontier]
            explored_list = list(explored)

            iter_str = f"{i:6}  "
            fringe_str = f"{str(frontier_list):<55}"[:55]
            explored_str = f"{str(explored_list):<50}"[:50]

            print(f"{iter_str}| {fringe_str} | {explored_str}")

            min_index = 0
            for j in range(1, len(frontier)):
                if frontier[j][1] < frontier[min_index][1]:
                    min_index = j

            path, f_cost = frontier.pop(min_index)
            current = path[-1]

            explored.append(current)

            if current == goal:
                print("-" * 110)
                return path, f_cost

            for child, w in self.adjacency_List[current]:
                if child not in explored:
                    new_f = f_cost + w + self.h[child] - self.h[current]
                    new_path = path + [child]

                    for k in range(len(frontier)):
                        if frontier[k][0][-1] == child and new_f < frontier[k][1]:
                            frontier[k] = (new_path, new_f)
                            break
                    else:
                        frontier.append((new_path, new_f))

            i += 1
